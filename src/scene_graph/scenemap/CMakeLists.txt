cmake_minimum_required(VERSION 3.18)
project(scenemap LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")

add_library(scenemap STATIC src/scan.cpp src/grid.cpp src/slam2d.cpp src/fk.cpp src/capi.cpp src/objmap.cpp src/bestview.cpp src/cloud.cpp src/png.cpp
            src/dsg_save.cpp src/rooms.cpp)
target_include_directories(scenemap PUBLIC include)
find_package(ZLIB REQUIRED)   # best view PNG(deflate·crc32)
target_link_libraries(scenemap PUBLIC ZLIB::ZLIB)
# Spark-DSG(저장, scene.json). 없으면 view.json·map.pgm 만 쓴다. 설치: refs/code/Spark-DSG → ~/.local (docs/Linux_설치.md)
list(APPEND CMAKE_PREFIX_PATH "$ENV{HOME}/.local")
find_package(spark_dsg QUIET)
if(spark_dsg_FOUND)
  target_link_libraries(scenemap PUBLIC spark_dsg::spark_dsg)
  target_compile_definitions(scenemap PRIVATE SM_HAVE_SPARK_DSG)
  message(STATUS "scenemap: Spark-DSG 저장 켬")
else()
  message(STATUS "scenemap: Spark-DSG 없음 — scene.json 없이 view.json·map.pgm 만")
endif()
set_target_properties(scenemap PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(scenemap PRIVATE -Wall -Wextra)

add_executable(slam2d_eval tools/slam2d_eval.cpp)
target_link_libraries(slam2d_eval scenemap)
add_executable(objmap_eval tools/objmap_eval.cpp)
target_link_libraries(objmap_eval scenemap)
add_executable(map_timeline tools/map_timeline.cpp)
target_link_libraries(map_timeline scenemap ZLIB::ZLIB)
# sgrt 기록(SGRT_RECORD) 재생: 자세 모드 비교·단계별 µs(tools/sm_bench.cpp)
add_executable(sm_bench tools/sm_bench.cpp)
target_link_libraries(sm_bench scenemap)
add_executable(capi_replay tools/capi_replay.cpp)
target_link_libraries(capi_replay scenemap)

enable_testing()
add_executable(test_fk tests/test_fk.cpp)
target_link_libraries(test_fk scenemap)
add_test(NAME fk COMMAND test_fk)
# 물체 기억(C ABI): 구조물·movable, 상자 이상값, 사라짐, best view, PNG·scene.json(spark_dsg 로 다시 읽음)
add_executable(test_objmem tests/test_objmem.cpp)
target_link_libraries(test_objmem scenemap ZLIB::ZLIB)
if(spark_dsg_FOUND)
  target_compile_definitions(test_objmem PRIVATE SM_TEST_SPARK_DSG)
endif()
add_test(NAME objmem COMMAND test_objmem ${CMAKE_CURRENT_BINARY_DIR}/test_objmem_out)
# 방 나누기(rooms.hpp): 두 방·ㄱ자·복도+방 셋·잡음·자람(id 유지)·물체 배정·이름·저장(ROOMS 층)·600×600 시간
add_executable(test_rooms tests/test_rooms.cpp)
target_link_libraries(test_rooms scenemap)
if(spark_dsg_FOUND)
  target_compile_definitions(test_rooms PRIVATE SM_TEST_SPARK_DSG)
endif()
add_test(NAME rooms COMMAND test_rooms ${CMAKE_CURRENT_BINARY_DIR}/test_rooms_out)
# 자세 원천(GT/SLAM/ODOM)·서 있을 때 장애물 생김/없어짐(사건 기반 넣기)·단계 시간
add_executable(test_posemap tests/test_posemap.cpp)
target_link_libraries(test_posemap scenemap)
add_test(NAME posemap COMMAND test_posemap)
add_executable(rooms_pgm tools/rooms_pgm.cpp)
target_link_libraries(rooms_pgm scenemap)
